   594→   628→   628→   628→def extract_yaml_frontmatter(content: str) -> dict:
   595→   629→   629→   629→    """Extrait le YAML frontmatter d'un contenu Markdown."""
   596→   630→   630→   630→    match = re.match(r'^---\s*\n(.*?)\n---', content, re.DOTALL)
   597→   631→   631→   631→    if not match:
   598→   632→   632→   632→        return {}
   599→   633→   633→   633→
   600→   634→   634→   634→    yaml_text = match.group(1)
   601→   635→   635→   635→    metadata = {}
   602→   636→   636→   636→    for line in yaml_text.split('\n'):
   603→   637→   637→   637→        line = line.strip()
   604→   638→   638→   638→        if ':' in line and not line.startswith('-'):
   605→   639→   639→   639→            key, _, value = line.partition(':')
   606→   640→   640→   640→            key = key.strip()
   607→   641→   641→   641→            value = value.strip()
   608→   642→   642→   642→            # Nettoyer les guillemets
   609→   643→   643→   643→            if value.startswith('"') and value.endswith('"'):
   610→   644→   644→   644→                value = value[1:-1]
   611→   645→   645→   645→            elif value.startswith("'") and value.endswith("'"):
   612→   646→   646→   646→                value = value[1:-1]
   613→   647→   647→   647→            # Nettoyer les listes
   614→   648→   648→   648→            if value.startswith('[') and value.endswith(']'):
   615→   649→   649→   649→                value = value[1:-1]
   616→   650→   650→   650→            # Nettoyer le > YAML
   617→   651→   651→   651→            if value.startswith('>'):
   618→   652→   652→   652→                value = value[1:].strip()
   619→   653→   653→   653→            metadata[key] = value
   620→   654→   654→   654→
   621→   655→   655→   655→    return metadata

   624→   658→   658→   658→def classify_skill(metadata: dict, content: str) -> str:
   625→   659→   659→   659→    """Classifie un skill selon son contenu.
   626→   660→   660→   660→
   627→   661→   661→   661→    Returns:
   628→   662→   662→   662→        'executable' — Protocole complet avec étapes et checklist

   630→   664→   664→   664→        'agent' — Définit un agent avec capacités spécifiques
   631→   665→   665→   665→    """
   632→   666→   666→   666→    content_lower = content.lower()
   633→   667→   667→   667→
   634→   668→   668→   668→    # Heuristiques pour classifier

   636→   670→   670→   670→    has_triggers = bool(re.search(r'déclencheur|declencheur|trigger|mot-clé|mot-cle', content_lower))
   637→   671→   671→   671→    has_agent = bool(re.search(r'agent spécialisé|agent specialise|agent.*capacité|capacit.*agent', content_lower))
   638→   672→   672→   672→
   639→   673→   673→   673→    if has_agent:
   640→   674→   674→   674→        return 'agent'
   641→   675→   675→   675→    elif has_steps or has_triggers:
   642→   676→   676→   676→        return 'executable'
   643→   677→   677→   677→    else:
