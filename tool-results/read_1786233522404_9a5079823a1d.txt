    96→   395→   594→   628→   628→   628→def extract_yaml_frontmatter(content: str) -> dict:
    97→   396→   595→   629→   629→   629→    """Extrait le YAML frontmatter d'un contenu Markdown."""
    98→   397→   596→   630→   630→   630→    match = re.match(r'^---\s*\n(.*?)\n---', content, re.DOTALL)
    99→   398→   597→   631→   631→   631→    if not match:
   100→   399→   598→   632→   632→   632→        return {}
   101→   400→   599→   633→   633→   633→
   102→   401→   600→   634→   634→   634→    yaml_text = match.group(1)
   103→   402→   601→   635→   635→   635→    metadata = {}
   104→   403→   602→   636→   636→   636→    for line in yaml_text.split('\n'):
   105→   404→   603→   637→   637→   637→        line = line.strip()
   106→   405→   604→   638→   638→   638→        if ':' in line and not line.startswith('-'):
   107→   406→   605→   639→   639→   639→            key, _, value = line.partition(':')
   108→   407→   606→   640→   640→   640→            key = key.strip()
   109→   408→   607→   641→   641→   641→            value = value.strip()
   110→   409→   608→   642→   642→   642→            # Nettoyer les guillemets
   111→   410→   609→   643→   643→   643→            if value.startswith('"') and value.endswith('"'):
   112→   411→   610→   644→   644→   644→                value = value[1:-1]
   113→   412→   611→   645→   645→   645→            elif value.startswith("'") and value.endswith("'"):
   114→   413→   612→   646→   646→   646→                value = value[1:-1]
   115→   414→   613→   647→   647→   647→            # Nettoyer les listes

   119→   418→   617→   651→   651→   651→            if value.startswith('>'):
   120→   419→   618→   652→   652→   652→                value = value[1:].strip()
   121→   420→   619→   653→   653→   653→            metadata[key] = value
